<?php
$input = '2, 3, 3, 3, 4, 8, 9, 1, 0, 1, 1, 3';
$frequencyText = '';
$uniqueText = '';
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
	$input = trim((string)($_POST['numbers'] ?? ''));
	$tokens = preg_split('/\s*,\s*/', $input);

	if ($input === '' || $tokens === false || count($tokens) === 0) {
		$error = 'Vui lòng nhập các số nguyên, cách nhau bằng dấu phẩy.';
	} else {
		$numbers = [];
		foreach ($tokens as $token) {
			if (!preg_match('/^-?\d+$/', $token)) {
				$error = 'Dữ liệu không hợp lệ. Chỉ nhập số nguyên và ngăn cách bằng dấu phẩy.';
				break;
			}
			$numbers[] = (int)$token;
		}

		if ($error === '') {
			$counts = array_count_values($numbers);
			$frequencyText = implode('  ', array_map(
				static fn($number, $count) => $number . ':' . $count,
				array_keys($counts),
				array_values($counts)
			));
			$uniqueText = implode(', ', array_keys($counts));
		}
	}
}

					id="numbers"
					name="numbers"

					value="<?= htmlspecialchars($input, ENT_QUOTES, 'UTF-8') ?>"

			</div>
			<div class="field">
				<label for="unique">Mảng duy nhất:</label>
				<input id="unique" type="text" value="<?= htmlspecialchars($uniqueText, ENT_QUOTES, 'UTF-8') ?>" readonly>
			</div>
			<div class="actions"><button type="submit">Thực hiện</button></div>
