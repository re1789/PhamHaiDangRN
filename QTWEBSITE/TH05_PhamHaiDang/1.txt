	$inputYear = filter_input(INPUT_POST, 'year', FILTER_VALIDATE_INT);
	if ($inputYear === false || $inputYear === null || $inputYear < 1 || $inputYear > 9999) {
		$error = 'Vui lòng nhập năm dương lịch từ 1 đến 9999.';
		$year = (string)($_POST['year'] ?? '');
	} else {
		$year = $inputYear;
	}
}

$validYear = $error === '';
if ($validYear) {
	$canIndex = ($year + 6) % 10;
	$chiIndex = ($year + 8) % 12;
	$lunarYear = $canNames[$canIndex] . ' ' . $chiNames[$chiIndex];
	$animalImage = '12congiap/' . $animalImages[$chiIndex];
}
